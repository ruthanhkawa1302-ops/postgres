## 2. Tạo bảng và index

Kiểu dữ liệu:

```sql
CREATE SCHEMA shop;

CREATE TYPE shop.order_status AS ENUM
  ('PENDING', 'PAID', 'PROCESSING', 'SHIPPED', 'DELIVERED', 'CANCELLED', 'RETURNED');

CREATE DOMAIN shop.money_amount AS numeric(12,2) CHECK (VALUE >= 0);
```
7 bảng: countries, categories, customers, products, orders, order_items, payments.

```sql
CREATE TABLE shop.countries (
    country_code  char(2)      PRIMARY KEY,
    country_name  varchar(100) NOT NULL,
    region        varchar(50)  NOT NULL,
    currency_code char(3)      NOT NULL,
    tax_rate      numeric(5,4) NOT NULL DEFAULT 0 CHECK (tax_rate BETWEEN 0 AND 1)
);

CREATE TABLE shop.categories (
    category_id        integer GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    parent_category_id integer REFERENCES shop.categories (category_id) ON DELETE SET NULL,
    category_name      varchar(100) NOT NULL,
    slug               varchar(120) NOT NULL
);

CREATE TABLE shop.customers (
    customer_id   bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    first_name    varchar(50)  NOT NULL,
    last_name     varchar(50)  NOT NULL,
    email         varchar(255) NOT NULL CHECK (email ~ '^[^@\s]+@[^@\s]+\.[^@\s]+$'),
    phone         varchar(20),
    gender        char(1) CHECK (gender IN ('M', 'F', 'O')),
    date_of_birth date,
    country_code  char(2) NOT NULL REFERENCES shop.countries (country_code),
    city          varchar(100) NOT NULL,
    loyalty_tier  varchar(10) NOT NULL DEFAULT 'BRONZE'
                  CHECK (loyalty_tier IN ('BRONZE', 'SILVER', 'GOLD', 'PLATINUM')),
    is_active     boolean NOT NULL DEFAULT true,
    created_at    timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE shop.products (
    product_id    bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    sku           varchar(30)  NOT NULL UNIQUE,
    product_name  varchar(200) NOT NULL,
    description   text,
    category_id   integer NOT NULL REFERENCES shop.categories (category_id),
    brand         varchar(60),
    unit_price    shop.money_amount NOT NULL CHECK (unit_price > 0),
    unit_cost     shop.money_amount NOT NULL,
    stock_qty     integer NOT NULL DEFAULT 0 CHECK (stock_qty >= 0),
    attributes    jsonb NOT NULL DEFAULT '{}'::jsonb,
    is_active     boolean NOT NULL DEFAULT true,
    created_at    timestamptz NOT NULL DEFAULT now(),
    updated_at    timestamptz NOT NULL DEFAULT now(),
    CONSTRAINT chk_products_margin CHECK (unit_cost <= unit_price)
);

CREATE TABLE shop.orders (
    order_id        bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    order_number    varchar(20) NOT NULL UNIQUE,
    customer_id     bigint NOT NULL REFERENCES shop.customers (customer_id),
    order_status    shop.order_status NOT NULL DEFAULT 'PENDING',
    sales_channel   varchar(10) NOT NULL DEFAULT 'WEB'
                    CHECK (sales_channel IN ('WEB', 'MOBILE', 'STORE', 'PARTNER')),
    order_date      timestamptz NOT NULL DEFAULT now(),
    subtotal        shop.money_amount NOT NULL DEFAULT 0,
    discount_amount shop.money_amount NOT NULL DEFAULT 0,
    tax_amount      shop.money_amount NOT NULL DEFAULT 0,
    shipping_fee    shop.money_amount NOT NULL DEFAULT 0,
    total_amount    shop.money_amount NOT NULL DEFAULT 0,
    currency_code   char(3) NOT NULL DEFAULT 'USD',
    updated_at      timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE shop.order_items (
    order_item_id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    order_id      bigint   NOT NULL REFERENCES shop.orders (order_id) ON DELETE CASCADE,
    line_no       smallint NOT NULL CHECK (line_no > 0),
    product_id    bigint   NOT NULL REFERENCES shop.products (product_id),
    quantity      integer  NOT NULL CHECK (quantity > 0),
    unit_price    shop.money_amount NOT NULL,
    discount_pct  numeric(5,2) NOT NULL DEFAULT 0 CHECK (discount_pct BETWEEN 0 AND 100),
    line_total    shop.money_amount NOT NULL,
    CONSTRAINT uq_order_items_order_line UNIQUE (order_id, line_no)
);

CREATE TABLE shop.payments (
    payment_id      bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    order_id        bigint NOT NULL REFERENCES shop.orders (order_id) ON DELETE CASCADE,
    payment_method  varchar(20) NOT NULL
                    CHECK (payment_method IN ('CREDIT_CARD', 'DEBIT_CARD', 'PAYPAL', 'BANK_TRANSFER', 'CASH_ON_DELIVERY')),
    payment_status  varchar(10) NOT NULL
                    CHECK (payment_status IN ('CAPTURED', 'FAILED', 'REFUNDED')),
    amount          shop.money_amount NOT NULL,
    transaction_ref varchar(40) NOT NULL,
    paid_at         timestamptz NOT NULL
);
```

Index. PK và UNIQUE ở trên đã tự tạo index, phần dưới tạo thêm mỗi loại một cái:

```sql
CREATE INDEX idx_order_items_product     ON shop.order_items (product_id);                -- B-tree trên FK
CREATE INDEX idx_payments_order          ON shop.payments (order_id);                     -- B-tree trên FK
CREATE INDEX idx_products_category_price ON shop.products (category_id, unit_price);      -- composite
CREATE INDEX idx_orders_customer_date    ON shop.orders (customer_id, order_date DESC)
       INCLUDE (order_status, total_amount);                                              -- covering
CREATE INDEX idx_orders_status_open      ON shop.orders (order_date)
       WHERE order_status IN ('PENDING', 'PAID', 'PROCESSING');                           -- partial
CREATE UNIQUE INDEX uq_customers_email_lower ON shop.customers (lower(email));            -- expression
CREATE INDEX idx_products_attributes_gin ON shop.products USING gin (attributes jsonb_path_ops);
CREATE INDEX idx_payments_paid_at_brin   ON shop.payments USING brin (paid_at) WITH (pages_per_range = 64);
CREATE INDEX idx_payments_txn_ref_hash   ON shop.payments USING hash (transaction_ref);
```


## 3. Sinh dữ liệu

Số dòng tính theo `scale`: customers = 500.000 × scale, products = 50.000 × scale, orders = 3.800.000 × scale.
Với 0.44 được khoảng 220k khách, 22k sản phẩm, 1,67 triệu đơn, 5 triệu dòng đơn hàng, 1,67 triệu thanh toán.

Dữ liệu sinh hoàn toàn bằng SQL (`generate_series` + `random()`, có `setseed` để lần nào chạy cũng ra giống nhau).
Tên người, thương hiệu, màu sắc... lấy ngẫu nhiên từ các mảng mẫu trong schema tạm `datagen`, chạy xong thì xóa.
Đầu script có `TRUNCATE ... RESTART IDENTITY CASCADE` nên chạy lại bao nhiêu lần cũng được.

Thứ tự nạp: countries (30 dòng cố định) → categories (5 gốc + 25 con) → customers → products → orders/order_items/payments.

Customers:

```sql
INSERT INTO shop.customers (customer_id, first_name, last_name, email, phone, gender, date_of_birth,
                            country_code, city, loyalty_tier, is_active, created_at)
SELECT g.id, g.fn, g.ln,
       lower(g.fn || '.' || g.ln || g.id || '@' || g.dom),
       '+' || (1 + g.id % 98) || '-' || lpad((g.id * 7919 % 10000000)::text, 7, '0'),
       (ARRAY['M','F','O'])[1 + (g.id % 3)::int],
       DATE '1950-01-01' + (random() * 20000)::int,
       c.country_code, c.city,
       CASE WHEN g.r < 0.55 THEN 'BRONZE' WHEN g.r < 0.85 THEN 'SILVER' WHEN g.r < 0.97 THEN 'GOLD' ELSE 'PLATINUM' END,
       random() > 0.03,
       timestamptz '2020-01-01' + interval '2190 days' * (g.id::float8 / current_setting('datagen.n_customers')::bigint)
FROM (
    SELECT id,
           datagen.pick(datagen.first_names())   AS fn,
           datagen.pick(datagen.last_names())    AS ln,
           datagen.pick(datagen.email_domains()) AS dom,
           1 + floor(random() * 38)::int         AS city_id,
           random()                              AS r
    FROM generate_series(1, current_setting('datagen.n_customers')::bigint) AS id
) g
JOIN datagen.cities c ON c.city_id = g.city_id;
```

Products: tên ghép từ thương hiệu + tính từ + loại hàng, `attributes` là JSON gồm color, material, size, warranty.

```sql
INSERT INTO shop.products (product_id, sku, product_name, description, category_id, brand,
                           unit_price, unit_cost, stock_qty, attributes, is_active, created_at, updated_at)
SELECT g.id,
       'SKU-' || lpad(g.id::text, 8, '0'),
       g.brand || ' ' || g.adj || ' ' || l.noun || ' ' || chr(65 + (g.id % 26)::int) || (100 + g.id % 900),
       g.adj || ' ' || lower(l.noun) || ' by ' || g.brand || '. '
         || datagen.pick(datagen.desc_phrases()) || ' ' || datagen.pick(datagen.desc_phrases()),
       5 + l.leaf_no,
       g.brand,
       g.price,
       round(g.price * (0.40 + random() * 0.40)::numeric, 2),
       CASE WHEN random() < 0.05 THEN 0 ELSE 10 + floor(random() * 990)::int END,
       jsonb_build_object(
           'color',           datagen.pick(datagen.colors()),
           'material',        datagen.pick(datagen.materials()),
           'size',            datagen.pick(datagen.sizes()),
           'warranty_months', (ARRAY[0, 6, 12, 24])[1 + (g.id % 4)::int]),
       random() > 0.05,
       g.created_at, g.created_at
FROM (
    SELECT id,
           datagen.pick(datagen.brands())     AS brand,
           datagen.pick(datagen.adjectives()) AS adj,
           1 + floor(random() * 25)::int      AS leaf_no,
           round((4.99 + power(random(), 3) * 1995)::numeric, 2) AS price,
           timestamptz '2020-01-01' + interval '1800 days' * (id::float8 / current_setting('datagen.n_products')::bigint) AS created_at
    FROM generate_series(1, current_setting('datagen.n_products')::bigint) AS id
) g
JOIN datagen.leaf l ON l.leaf_no = g.leaf_no;
```

Orders, order_items, payments được nạp bằng procedure `datagen.load_orders()`, mỗi lô 250.000 đơn rồi COMMIT.
Trong mỗi lô:

1. Sinh 1–5 dòng hàng cho mỗi đơn vào bảng tạm `tmp_items`. Sản phẩm chọn bằng `power(random(), 2)` để có hàng bán chạy, hàng bán ế.
2. Insert `orders`, tổng tiền tính từ `tmp_items` (thuế 8%, miễn phí ship từ 100 USD). `order_date` tăng dần theo id từ 2021 đến 2025.
3. Insert `order_items` từ `tmp_items`.
4. Insert `payments`: mỗi đơn khác PENDING có 1 thanh toán, 4% đơn có thêm 1 lần FAILED.

```sql
CALL datagen.load_orders();
```

Cuối script xóa schema `datagen` và chạy `VACUUM (ANALYZE)` các bảng.

### Sequence, type, aggregate

```sql
CREATE SEQUENCE shop.order_number_seq AS bigint;     -- sinh số đơn cho đơn mới
SELECT setval('shop.order_number_seq', (SELECT coalesce(max(order_id), 0) + 1 FROM shop.orders), false);

-- Trung bình có trọng số: weighted_avg(giá trị, trọng số)
CREATE OR REPLACE FUNCTION shop.wavg_sfunc(state numeric[], val numeric, weight numeric)
RETURNS numeric[] LANGUAGE sql IMMUTABLE PARALLEL SAFE AS $$
    SELECT CASE WHEN val IS NULL OR weight IS NULL THEN state
                ELSE ARRAY[state[1] + val * weight, state[2] + weight] END
$$;
CREATE OR REPLACE FUNCTION shop.wavg_ffunc(state numeric[])
RETURNS numeric LANGUAGE sql IMMUTABLE PARALLEL SAFE AS $$
    SELECT CASE WHEN state[2] = 0 THEN NULL ELSE round(state[1] / state[2], 4) END
$$;
CREATE OR REPLACE AGGREGATE shop.weighted_avg(numeric, numeric) (
    SFUNC = shop.wavg_sfunc, STYPE = numeric[], FINALFUNC = shop.wavg_ffunc, INITCOND = '{0,0}'
);
```

### Materialized view

Doanh thu theo ngày và kênh bán. Phải tạo trước các function đọc từ nó.

```sql
CREATE MATERIALIZED VIEW shop.mv_daily_sales AS
SELECT (o.order_date AT TIME ZONE 'UTC')::date AS sales_date,
       o.sales_channel,
       count(*)                       AS orders_count,
       sum(o.total_amount)            AS revenue,
       round(avg(o.total_amount), 2)  AS avg_order_value,
       count(DISTINCT o.customer_id)  AS unique_customers
FROM shop.orders o
WHERE o.order_status <> 'CANCELLED'
GROUP BY 1, 2;

CREATE UNIQUE INDEX uq_mv_daily_sales ON shop.mv_daily_sales (sales_date, sales_channel);
-- làm mới: REFRESH MATERIALIZED VIEW CONCURRENTLY shop.mv_daily_sales;
```

### Function

Tổng chi tiêu của một khách:

```sql
CREATE OR REPLACE FUNCTION shop.fn_customer_lifetime_value(p_customer_id bigint)
RETURNS numeric LANGUAGE plpgsql STABLE AS $$
DECLARE
    v_total numeric;
BEGIN
    IF NOT EXISTS (SELECT 1 FROM shop.customers WHERE customer_id = p_customer_id) THEN
        RAISE EXCEPTION 'Customer % does not exist', p_customer_id USING ERRCODE = 'no_data_found';
    END IF;

    SELECT coalesce(sum(total_amount), 0) INTO v_total
    FROM shop.orders
    WHERE customer_id = p_customer_id
      AND order_status NOT IN ('CANCELLED', 'RETURNED');

    RETURN v_total;
END;
$$;
```

Chi tiết một đơn hàng dạng JSON:

```sql
CREATE OR REPLACE FUNCTION shop.fn_get_order_details(p_order_id bigint)
RETURNS jsonb LANGUAGE sql STABLE AS $$
    SELECT jsonb_build_object(
        'order',    to_jsonb(o) - 'customer_id',
        'customer', jsonb_build_object('customer_id', c.customer_id,
                                       'name', c.first_name || ' ' || c.last_name,
                                       'email', c.email, 'tier', c.loyalty_tier),
        'items',    (SELECT jsonb_agg(jsonb_build_object('line_no', i.line_no, 'sku', p.sku,
                                                         'product', p.product_name, 'quantity', i.quantity,
                                                         'unit_price', i.unit_price, 'line_total', i.line_total)
                                      ORDER BY i.line_no)
                     FROM shop.order_items i JOIN shop.products p ON p.product_id = i.product_id
                     WHERE i.order_id = o.order_id),
        'payments', (SELECT jsonb_agg(to_jsonb(pm) - 'order_id' ORDER BY pm.paid_at)
                     FROM shop.payments pm WHERE pm.order_id = o.order_id))
    FROM shop.orders o
    JOIN shop.customers c ON c.customer_id = o.customer_id
    WHERE o.order_id = p_order_id
$$;
```

Tìm sản phẩm gần đúng theo tên (pg_trgm):

```sql
CREATE OR REPLACE FUNCTION shop.fn_search_products(p_query text, p_limit int DEFAULT 20)
RETURNS TABLE (product_id bigint, sku text, product_name text, unit_price numeric, score real)
LANGUAGE sql STABLE SET search_path = shop, public AS $$
    SELECT p.product_id, p.sku::text, p.product_name::text, p.unit_price::numeric,
           similarity(p.product_name, p_query) AS score
    FROM shop.products p
    WHERE p.product_name % p_query
       OR p.product_name ILIKE '%' || p_query || '%'
    ORDER BY score DESC, p.product_id
    LIMIT p_limit
$$;
```

Doanh thu 12 tháng theo kênh bán (tablefunc):

```sql
CREATE OR REPLACE FUNCTION shop.fn_monthly_sales_pivot(p_year int)
RETURNS TABLE (sales_month int, web numeric, mobile numeric, store numeric, partner numeric)
LANGUAGE sql STABLE AS $$
    SELECT *
    FROM public.crosstab(
        format($q$
            SELECT extract(month FROM sales_date)::int, sales_channel, sum(revenue)
            FROM shop.mv_daily_sales
            WHERE sales_date >= make_date(%1$s, 1, 1) AND sales_date < make_date(%1$s + 1, 1, 1)
            GROUP BY 1, 2 ORDER BY 1, 2 $q$, p_year),
        $c$ VALUES ('WEB'), ('MOBILE'), ('STORE'), ('PARTNER') $c$
    ) AS ct(sales_month int, web numeric, mobile numeric, store numeric, partner numeric)
$$;
```

Sản phẩm cùng danh mục có tên giống nhất (pg_trgm):

```sql
CREATE OR REPLACE FUNCTION shop.fn_similar_products(p_product_id bigint, p_limit int DEFAULT 10)
RETURNS TABLE (product_id bigint, product_name text, category_id int, score real)
LANGUAGE plpgsql STABLE SET search_path = shop, public AS $$
DECLARE
    v_name     text;
    v_category int;
BEGIN
    SELECT p.product_name, p.category_id INTO v_name, v_category
    FROM shop.products p WHERE p.product_id = p_product_id;
    IF NOT FOUND THEN
        RAISE EXCEPTION 'Product % does not exist', p_product_id;
    END IF;

    RETURN QUERY
    SELECT p.product_id, p.product_name::text, p.category_id, similarity(p.product_name, v_name)
    FROM shop.products p
    WHERE p.category_id = v_category
      AND p.product_id <> p_product_id
    ORDER BY 4 DESC, p.product_id
    LIMIT p_limit;
END;
$$;
```

Các kho gần một tọa độ nhất (công thức haversine):

```sql
CREATE OR REPLACE FUNCTION shop.fn_nearest_warehouses(p_lat double precision, p_lon double precision, p_limit int DEFAULT 3)
RETURNS TABLE (warehouse_code text, city text, distance_km numeric)
LANGUAGE sql STABLE AS $$
    SELECT w.warehouse_code::text, w.city::text,
           round((2 * 6371 * asin(sqrt(
                 sin(radians(w.location[1] - p_lat) / 2) ^ 2
               + cos(radians(p_lat)) * cos(radians(w.location[1]))
                 * sin(radians(w.location[0] - p_lon) / 2) ^ 2)))::numeric, 1) AS distance_km
    FROM shop.warehouses w
    ORDER BY distance_km
    LIMIT p_limit
$$;
```

### Trigger

`updated_at` tự cập nhật khi sửa đơn, và ghi audit khi sửa/xóa orders, products (chỉ lưu các cột thay đổi).

```sql
CREATE OR REPLACE FUNCTION shop.trg_set_updated_at()
RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
    NEW.updated_at := now();
    RETURN NEW;
END;
$$;

CREATE OR REPLACE FUNCTION shop.trg_audit_row()
RETURNS trigger LANGUAGE plpgsql SET search_path = shop, public AS $$
DECLARE
    v_diff hstore;
BEGIN
    IF TG_OP = 'UPDATE' THEN
        v_diff := (hstore(NEW) - hstore(OLD)) - ARRAY['updated_at'];
        IF v_diff = ''::hstore THEN
            RETURN NULL;
        END IF;
        INSERT INTO shop.audit_log (table_name, operation, row_pk, changed_fields)
        VALUES (TG_TABLE_NAME, TG_OP, hstore(OLD) -> TG_ARGV[0], v_diff);
    ELSE
        INSERT INTO shop.audit_log (table_name, operation, row_pk, detail)
        VALUES (TG_TABLE_NAME, TG_OP, hstore(OLD) -> TG_ARGV[0], to_jsonb(OLD));
    END IF;
    RETURN NULL;
END;
$$;

CREATE OR REPLACE TRIGGER trg_orders_updated_at
    BEFORE UPDATE ON shop.orders
    FOR EACH ROW EXECUTE FUNCTION shop.trg_set_updated_at();
CREATE OR REPLACE TRIGGER trg_orders_audit
    AFTER UPDATE OR DELETE ON shop.orders
    FOR EACH ROW EXECUTE FUNCTION shop.trg_audit_row('order_id');
CREATE OR REPLACE TRIGGER trg_products_audit
    AFTER UPDATE OF unit_price, unit_cost, is_active OR DELETE ON shop.products
    FOR EACH ROW EXECUTE FUNCTION shop.trg_audit_row('product_id');
```

```sql
CREATE OR REPLACE FUNCTION shop.trg_log_ddl()
RETURNS event_trigger LANGUAGE plpgsql SECURITY DEFINER SET search_path = pg_catalog, shop AS $$
DECLARE
    r record;
BEGIN
    FOR r IN SELECT * FROM pg_event_trigger_ddl_commands() LOOP
        INSERT INTO shop.audit_log (table_name, operation, detail)

                jsonb_build_object('command_tag', r.command_tag, 'object_type', r.object_type));
    END LOOP;
END;
$$;

CREATE EVENT TRIGGER evt_ddl_audit ON ddl_command_end
```

### Procedure

Tạo đơn mới: trừ tồn kho, thêm dòng hàng, tính tổng tiền.

```sql
CREATE OR REPLACE PROCEDURE shop.sp_place_order(
    p_customer_id bigint,
    p_items       shop.order_line_input[],
    INOUT p_order_id bigint DEFAULT NULL)
LANGUAGE plpgsql AS $$
DECLARE
    v_item  shop.order_line_input;
    v_line  smallint := 0;
    v_price numeric;
BEGIN
    IF p_items IS NULL OR cardinality(p_items) = 0 THEN
        RAISE EXCEPTION 'Order must contain at least one item';
    END IF;

    INSERT INTO shop.orders (order_number, customer_id, order_status, sales_channel)
    VALUES ('ORD-' || lpad(nextval('shop.order_number_seq')::text, 10, '0'), p_customer_id, 'PENDING', 'WEB')
    RETURNING order_id INTO p_order_id;

    FOREACH v_item IN ARRAY p_items LOOP
        v_line := v_line + 1;

        UPDATE shop.products
           SET stock_qty = stock_qty - v_item.quantity
         WHERE product_id = v_item.product_id
           AND is_active
           AND stock_qty >= v_item.quantity
        RETURNING unit_price INTO v_price;
        IF NOT FOUND THEN
            RAISE EXCEPTION 'Product % is inactive or out of stock', v_item.product_id;
        END IF;

        INSERT INTO shop.order_items (order_id, line_no, product_id, quantity, unit_price, line_total)
        VALUES (p_order_id, v_line, v_item.product_id, v_item.quantity, v_price, v_item.quantity * v_price);
    END LOOP;

    UPDATE shop.orders o
       SET subtotal     = s.net,
           tax_amount   = round(s.net * 0.08, 2),
           shipping_fee = CASE WHEN s.net >= 100 THEN 0 ELSE 4.99 END,
           total_amount = s.net + round(s.net * 0.08, 2) + CASE WHEN s.net >= 100 THEN 0 ELSE 4.99 END
      FROM (SELECT sum(line_total) AS net FROM shop.order_items WHERE order_id = p_order_id) s
     WHERE o.order_id = p_order_id;

    RAISE NOTICE 'Order % created for customer %', p_order_id, p_customer_id;
END;
$$;
```

Đổi trạng thái đơn, chỉ cho phép theo luồng PENDING → PAID → PROCESSING → SHIPPED → DELIVERED → RETURNED
(được hủy khi chưa SHIPPED).

```sql
CREATE OR REPLACE PROCEDURE shop.sp_update_order_status(p_order_id bigint, p_new_status shop.order_status)
LANGUAGE plpgsql AS $$
DECLARE
    v_current shop.order_status;
BEGIN
    SELECT order_status INTO v_current FROM shop.orders WHERE order_id = p_order_id FOR UPDATE;
    IF NOT FOUND THEN
        RAISE EXCEPTION 'Order % not found', p_order_id;
    END IF;

    IF NOT (   (v_current = 'PENDING'    AND p_new_status IN ('PAID', 'CANCELLED'))
            OR (v_current = 'PAID'       AND p_new_status IN ('PROCESSING', 'CANCELLED'))
            OR (v_current = 'PROCESSING' AND p_new_status IN ('SHIPPED', 'CANCELLED'))
            OR (v_current = 'SHIPPED'    AND p_new_status =  'DELIVERED')
            OR (v_current = 'DELIVERED'  AND p_new_status =  'RETURNED')) THEN
        RAISE EXCEPTION 'Invalid status transition % -> % for order %', v_current, p_new_status, p_order_id;
    END IF;

    UPDATE shop.orders SET order_status = p_new_status WHERE order_id = p_order_id;
END;
$$;
```

### View

```sql
-- Đơn hàng kèm tên khách, số dòng, trạng thái thanh toán gần nhất
CREATE VIEW shop.v_order_summary AS
SELECT o.order_id, o.order_number, o.order_date, o.order_status, o.sales_channel, o.total_amount,
       c.customer_id, c.first_name || ' ' || c.last_name AS customer_name, c.country_code,
       (SELECT count(*) FROM shop.order_items i WHERE i.order_id = o.order_id) AS item_count,
       lp.payment_status AS last_payment_status
FROM shop.orders o
JOIN shop.customers c ON c.customer_id = o.customer_id
LEFT JOIN LATERAL (SELECT p.payment_status FROM shop.payments p
                   WHERE p.order_id = o.order_id ORDER BY p.paid_at DESC LIMIT 1) lp ON true;

-- Khách hàng đã che email, số điện thoại
CREATE VIEW shop.v_customers_masked AS
SELECT customer_id,
       regexp_replace(email, '^(.)[^@]*(@.*)$', '\1***\2')            AS email_masked,
       encode(public.digest(lower(email), 'sha256'), 'hex')           AS email_sha256,
       left(phone, 4) || '*******'                                     AS phone_masked,
       country_code, city, loyalty_tier
FROM shop.customers;

-- Sản phẩm đang bán, update qua view không được làm sản phẩm thành inactive
CREATE VIEW shop.v_active_products AS
SELECT product_id, sku, product_name, category_id, brand, unit_price, unit_cost, stock_qty, is_active
FROM shop.products
WHERE is_active
WITH CASCADED CHECK OPTION;
```

### Role và phân quyền

`shop_readonly` đọc được mọi bảng trừ audit_log, `demo_analyst` là user đăng nhập thuộc nhóm đó.
`shop_portal` chỉ thấy đơn hàng của khách có id bằng `app.customer_id` (row level security).

```sql
CREATE ROLE shop_readonly NOLOGIN;
CREATE ROLE shop_portal   NOLOGIN;
CREATE ROLE demo_analyst  LOGIN PASSWORD 'Analyst@123' IN ROLE shop_readonly;

GRANT USAGE ON SCHEMA shop TO shop_readonly, shop_portal;
GRANT SELECT ON ALL TABLES IN SCHEMA shop TO shop_readonly;
GRANT EXECUTE ON ALL FUNCTIONS IN SCHEMA shop TO shop_readonly;
ALTER DEFAULT PRIVILEGES IN SCHEMA shop GRANT SELECT ON TABLES TO shop_readonly;
REVOKE SELECT ON shop.audit_log FROM shop_readonly;
GRANT SELECT ON shop.orders TO shop_portal;

ALTER TABLE shop.orders ENABLE ROW LEVEL SECURITY;
CREATE POLICY p_orders_readonly ON shop.orders
    FOR SELECT TO shop_readonly USING (true);
CREATE POLICY p_orders_portal ON shop.orders
    FOR SELECT TO shop_portal
    USING (customer_id = nullif(current_setting('app.customer_id', true), '')::bigint);
```
